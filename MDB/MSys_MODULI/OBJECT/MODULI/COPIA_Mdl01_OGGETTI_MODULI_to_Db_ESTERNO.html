' ====================================================================================
' SUB: EsportaOggettiGenericiInDBEsterno
' Esporta gli oggetti definiti in GetOggettiPerTrattamento verso un database esterno
' ====================================================================================
Public Sub EsportaOggettiGenericiInDBEsterno()
    On Error GoTo GestioneErrore

    Dim percorsoDestinazione As String
    percorsoDestinazione = "c:\Casa\LINGUAGGI\ACCESS\PROGETTI_MDB\MSYS_OGGETTI\MSYS\MDB\TMP_IMPORTA_MODULI\MSYS_MODELLO.mdb"

    ' Chiudo form se aperta
    DoCmd.Close acForm, "UTILITA_MsysDF14Frm05_ATTIVA_FUNZIONI"

    ' Ottengo lista oggetti
    Dim oggettiToExport As Collection
    Set oggettiToExport = GetOggettiPerTrattamento()

    If oggettiToExport.count = 0 Then
        MsgBox "Nessun oggetto da esportare.", vbInformation, "Esportazione"
        Exit Sub
    End If

    Dim item As Variant
    Dim tipoOggetto As AcObjectType
    Dim msg As String
    Dim countMacro As Long, countModuli As Long, countQuery As Long, countForm As Long, countReport As Long
    Dim NomeOrigine As String, NomeDestinazione As String, NomeDuplicato As String

    ' DEBUG: mostra lista prima di procedere
    Debug.Print "=== LISTA OGGETTI DA ESPORTARE ==="
    For Each item In oggettiToExport
        If IsArray(item) Then
            Debug.Print "Tipo:", item(0), "Orig:", item(1), "Dest:", item(2)
        Else
            Debug.Print "Elemento NON array:", TypeName(item)
        End If
    Next

    ' Ciclo di esportazione
    For Each item In oggettiToExport

        ' Sicurezza: l'elemento deve essere un array di almeno 3 elementi
        If Not IsArray(item) Then GoTo ProssimoItem
        If UBound(item) < 2 Then GoTo ProssimoItem

        ' Controllo campi obbligatori
        If Trim(Nz(item(0), "")) = "" Then GoTo ProssimoItem
        If Trim(Nz(item(1), "")) = "" Then GoTo ProssimoItem
        If Trim(Nz(item(2), "")) = "" Then GoTo ProssimoItem

        ' Tipo oggetto
        Select Case UCase(item(0))
            Case "MACRO":  tipoOggetto = acMacro:  countMacro = countMacro + 1
            Case "MODULE": tipoOggetto = acModule: countModuli = countModuli + 1
            Case "QUERY":  tipoOggetto = acQuery:  countQuery = countQuery + 1
            Case "FORM":   tipoOggetto = acForm:   countForm = countForm + 1
            Case "REPORT": tipoOggetto = acReport: countReport = countReport + 1
            Case Else
                MsgBox "Tipo oggetto non supportato: " & item(0), vbExclamation
                GoTo ProssimoItem
        End Select

        'controllo
           Debug.Print "oggetto esportato" & item(1)
           
           
           
        ' Caso speciale: duplico la form principale
        If item(1) = "UTILITA_MsysDF14Frm05_ATTIVA_FUNZIONI" Then
            NomeOrigine = item(1)
            NomeDuplicato = item(1) & "_duplicato"
            DoCmd.CopyObject , NomeDuplicato, acForm, NomeOrigine
            DoCmd.CopyObject percorsoDestinazione, NomeDuplicato, tipoOggetto, NomeDuplicato
        Else
            ' Copia normale
            DoCmd.CopyObject percorsoDestinazione, item(2), tipoOggetto, item(1)
        End If

ProssimoItem:
    Next item

    ' Riepilogo
    msg = "Esportazione completata." & vbCrLf & vbCrLf & _
          "Database di destinazione:" & vbCrLf & percorsoDestinazione & vbCrLf & vbCrLf & _
          "Oggetti esportati:" & vbCrLf & _
          "- Macro: " & countMacro & vbCrLf & _
          "- Moduli: " & countModuli & vbCrLf & _
          "- Query: " & countQuery & vbCrLf & _
          "- Form: " & countForm & vbCrLf & _
          "- Report: " & countReport & vbCrLf

    MsgBox msg, vbInformation, "Esportazione Oggetti Access"
    Exit Sub

GestioneErrore:
    MsgBox "Errore " & Err.Number & ": " & Err.Description, vbCritical, "Errore durante l'esportazione"
End Sub


' ====================================================================================
' Funzione wrapper per cancellazione
' ====================================================================================
Public Function ATTIVA_CANCELLAOggettiGenericiInDBEsterno()
    MsgBox "ATTIVATA FUNZIONE CANCELLA OGGETTI NEL PROGETTO.MDB", vbInformation
    Call CancellaOggettiGenericiInDBEsterno
End Function


' ====================================================================================
' SUB: CancellaOggettiGenericiInDBEsterno
' Cancella dal DB esterno gli oggetti elencati in GetOggettiPerTrattamento
' ====================================================================================
Public Sub CancellaOggettiGenericiInDBEsterno()
    On Error GoTo GestioneErrore

    Dim percorsoDestinazione As String
    percorsoDestinazione = "c:\Casa\LINGUAGGI\ACCESS\PROGETTI_MDB\MSYS_OGGETTI\MSYS\MDB\TMP_IMPORTA_MODULI\MSYS_MODELLO.mdb"

    DoCmd.Close acForm, "UTILITA_MsysDF14Frm05_ATTIVA_FUNZIONI"

    Dim oggettiToDelete As Collection
    Set oggettiToDelete = GetOggettiPerTrattamento()

    If oggettiToDelete.count = 0 Then
        MsgBox "Nessun oggetto da cancellare.", vbInformation, "Cancellazione"
        Exit Sub
    End If

    Dim accApp As Object
    Set accApp = CreateObject("Access.Application")
    accApp.OpenCurrentDatabase percorsoDestinazione

    Dim item As Variant
    Dim tipoOggetto As AcObjectType
    Dim msg As String
    Dim countMacro As Long, countModuli As Long, countQuery As Long, countForm As Long, countReport As Long

    ' DEBUG lista oggetti da cancellare
    Debug.Print "=== LISTA OGGETTI DA CANCELLARE ==="
    For Each item In oggettiToDelete
        If IsArray(item) Then
            Debug.Print "Tipo:", item(0), "Dest:", item(2)
        End If
    Next

    ' Cancellazione
    For Each item In oggettiToDelete
        If Not IsArray(item) Then GoTo ProssimoItem
        If UBound(item) < 2 Then GoTo ProssimoItem

        If Trim(Nz(item(0), "")) = "" Then GoTo ProssimoItem
        If Trim(Nz(item(2), "")) = "" Then GoTo ProssimoItem

        Select Case UCase(item(0))
            Case "MACRO":  tipoOggetto = acMacro:  countMacro = countMacro + 1
            Case "MODULE": tipoOggetto = acModule: countModuli = countModuli + 1
            Case "QUERY":  tipoOggetto = acQuery:  countQuery = countQuery + 1
            Case "FORM":   tipoOggetto = acForm:   countForm = countForm + 1
            Case "REPORT": tipoOggetto = acReport: countReport = countReport + 1
            Case Else
                GoTo ProssimoItem
        End Select

        accApp.DoCmd.DeleteObject tipoOggetto, item(2)

ProssimoItem:
    Next item

    accApp.CloseCurrentDatabase
    accApp.Quit
    Set accApp = Nothing

    msg = "Cancellazione completata." & vbCrLf & vbCrLf & _
          "Database di destinazione:" & vbCrLf & percorsoDestinazione & vbCrLf & vbCrLf & _
          "Oggetti cancellati:" & vbCrLf & _
          "- Macro: " & countMacro & vbCrLf & _
          "- Moduli: " & countModuli & vbCrLf & _
          "- Query: " & countQuery & vbCrLf & _
          "- Form: " & countForm & vbCrLf & _
          "- Report: " & countReport

    MsgBox msg, vbInformation, "Cancellazione Oggetti Access"
    Exit Sub

GestioneErrore:
    MsgBox "Errore " & Err.Number & ": " & Err.Description, vbCritical, "Errore durante la cancellazione"
End Sub
