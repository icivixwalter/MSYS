Option Compare Database
Option Explicit

Public Function ATTIVA_EsportaOggettiGenericiInDBEsterno()
    MsgBox "ATTIVATO LA FUNZIONE COPIA OGGETTI NEL PROGETTO.MDB", vbInformation, "ATTIVO COPIA OGGETTI NEL FILE ESTERNO"
        
    Call EsportaOggettiGenericiInDBEsterno
End Function

'//ESPORTA OGGETTI NEL DATABASE ESTERNO
'//------------------------------------------------------------------------------------------//
'//Note: per ora questa procedura esporta solo con dati fissi nel futuro puo essere implementata
'//      per l'esportazione dei dati presalvati in una tabella.

Public Sub EsportaOggettiGenericiInDBEsterno()
    On Error GoTo GestioneErrore

    ' === Percorso database di destinazione ===
    Dim percorsoDestinazione As String
    percorsoDestinazione = "c:\Casa\LINGUAGGI\ACCESS\PROGETTI_MDB\MSYS_OGGETTI\MSYS\MDB\TMP_IMPORTA_MODULI\MSYS_MODELLO.mdb"

    ' === Collection: Array("TipoOggetto", "NomeOrigine", "NomeDestinazione") ===
    Dim oggettiToExport As Collection
    Set oggettiToExport = New Collection

    ' -- Macro --
    oggettiToExport.Add Array("Macro", "SALVA_Mcr01_}--------------------------------------------------@", "SALVA_Mcr01_}--------------------------------------------------@")
    oggettiToExport.Add Array("Macro", "SALVA_Mcr01_TUTTI_GLI_OGGETTI_IN_HTML", "SALVA_Mcr01_TUTTI_GLI_OGGETTI_IN_HTML")
    oggettiToExport.Add Array("Macro", "", "")
    oggettiToExport.Add Array("Macro", "", "")

    ' -- Moduli --
    oggettiToExport.Add Array("Module", "UTILITA_MsysDF14Mdl05_}----------------------------------------@", "UTILITA_MsysDF14Mdl05_}----------------------------------------@")
    oggettiToExport.Add Array("Module", "UTILITA_MsysDF14Mdl05_IMPORTAZIONE_SEMPLICE_DI_OGGETTI", "UTILITA_MsysDF14Mdl05_IMPORTAZIONE_SEMPLICE_DI_OGGETTI")


    ' -- FORM --
    oggettiToExport.Add Array("Form", "UTILITA_MsysDF14Frm00_}----------------------------------------@", "UTILITA_MsysDF14Frm00_}----------------------------------------@")
    oggettiToExport.Add Array("Form", "UTILITA_MsysDF14Frm05_ATTIVA_FUNZIONI", "UTILITA_MsysDF14Frm05_ATTIVA_FUNZIONI")


    ' === Esecuzione dell’esportazione ===
    Dim item As Variant
    Dim tipoOggetto As AcObjectType
    Dim msg As String
    Dim countMacro As Long, countModuli As Long, countQuery As Long, countForm As Long, countReport As Long

    For Each item In oggettiToExport
        Select Case UCase(item(0))
            Case "MACRO"
                tipoOggetto = acMacro
                countMacro = countMacro + 1
            Case "MODULE"
                tipoOggetto = acModule
                countModuli = countModuli + 1
            Case "QUERY"
                tipoOggetto = acQuery
                countQuery = countQuery + 1
            Case "FORM"
                tipoOggetto = acForm
                countForm = countForm + 1
            Case "REPORT"
                tipoOggetto = acReport
                countReport = countReport + 1
            Case Else
                MsgBox "Tipo oggetto non supportato: " & item(0), vbExclamation
                GoTo GestioneErrore
        End Select

        ' === Uso corretto di CopyObject con parametri posizionali ===
        ' Syntax: CopyObject(destinationdatabase, newname, objecttype, sourceobjectname)
        DoCmd.CopyObject percorsoDestinazione, item(2), tipoOggetto, item(1)
    Next item

    ' === Riepilogo finale ===
    msg = "Esportazione completata." & vbCrLf & vbCrLf & _
          "Database di destinazione:" & vbCrLf & percorsoDestinazione & vbCrLf & vbCrLf & _
          "Oggetti esportati:" & vbCrLf & _
          "- Macro: " & countMacro & vbCrLf & _
          "- Moduli: " & countModuli & vbCrLf & _
          "- Query: " & countQuery & vbCrLf & _
          "- Form: " & countForm & vbCrLf & _
          "- Report: " & countReport & vbCrLf & vbCrLf & _
          "Dettaglio esportazioni:" & vbCrLf

    For Each item In oggettiToExport
        msg = msg & "• " & item(0) & ": " & item(1) & " ? " & item(2) & vbCrLf
    Next item

    MsgBox msg, vbInformation, "Esportazione Oggetti Access"
    Exit Sub

GestioneErrore:
    MsgBox "Errore " & Err.Number & ": " & Err.Description, vbCritical, "Errore durante l'esportazione"
End Sub
