# QUERY UPDATE NRO PROGETTO

## note con la query
 MSys_TB41_UPDATE_Qry01_01_NRO_PROG_i
 aggiorno il numero di progetto nella tabella tb41




 ## sql
   UPDATE MSys_TAB40_PROG_MASTER INNER JOIN MSys_TB41_VISUAL_PAG_MASTER ON MSys_TAB40_PROG_MASTER.COD_PROGETTO_s = MSys_TB41_VISUAL_PAG_MASTER.COD_PROGETTO_s SET MSys_TB41_VISUAL_PAG_MASTER.NRO_PROG_i = [MSys_TAB40_PROG_MASTER].[NRO_PROG_i]
                  WITH OWNERACCESS OPTION;
