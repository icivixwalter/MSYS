'// ESPORTA OGGETTI IN LLPP_ATTI_GESTIONE
'//===================================================================================================//
'//Note: esporto gli oggetti da Database di origine a quello di destinazione: _
        MSys_GE_FILE.mdb ---> LLPP_ATTI_GESTIONE.mdb

Public Function EsportaOggettiStandard_LLPP_ATTI_GESTIONE()
    On Error GoTo Esporta_Err

    ' === Impostazioni percorso database di destinazione ===
    Dim dbDestPath As String
    Dim dbDestFile As String
    
    dbDestPath = "C:\GESTIONI\GESTIONE_LLPP\25_GESTIONE_LLPP\LLPP_ARCHIVI_MDB\"
    dbDestFile = "LLPP_ATTI_GESTIONE.mdb"

    ' === Verifica se ci troviamo già nel database di destinazione ===
    If LCase(GetFileName(CurrentDb.Name)) = LCase(GetFileName(dbDestFile)) Then
        MsgBox "Sei già nel database di destinazione (" & dbDestFile & "). Operazione annullata.", vbExclamation
        Exit Function
    End If

    ' === Recupera oggetti da esportare ===
    Dim oggetti As Variant
    oggetti = GetOggettiDaEsportare()

    Dim i As Long
    Dim nomeOggetto As String
    Dim tipoOggetto As AcObjectType

    ' === Ciclo di esportazione ===
    For i = LBound(oggetti) To UBound(oggetti)
        nomeOggetto = oggetti(i)(0)
        tipoOggetto = oggetti(i)(1)

        ' Cancella l'oggetto nel database di destinazione (se esiste)
        CancellaOggettoNelDBEsterno dbDestPath, nomeOggetto, tipoOggetto

        ' Copia l’oggetto nel database di destinazione
        DoCmd.CopyObject dbDestPath, nomeOggetto, tipoOggetto, nomeOggetto
    Next i

    MsgBox "Esportazione oggetti completata.", vbInformation

Esporta_Exit:
    Exit Function

Esporta_Err:
    MsgBox "Errore durante l'esportazione: " & Err.Description, vbCritical
    Resume Esporta_Exit
End Function

'// REINSPORTA OGGETTI GEFILE IN ARCHIVIO ORIGINALE
'//===================================================================================================//
'// Riassunto : _
    Le funzioni (attivabili da una macro) permettono di esportare i moduli, le query e _
    le maschere (form) dal database ospitante (quello in cui lavori adesso) verso il database sorgente/madre (l’archivio originale). _
    Prima di copiare ogni oggetto nel database sorgente, se esiste già viene cancellato (solo per moduli, query e form), per evitare duplicati o conflitti. _
    La funzione blocca l’esecuzione se ti trovi già nel database sorgente, così eviti di esportare su te stesso e causare errori o duplicazioni inutili.
    
'//OCCORRONO TRE PROCEDURE: _
    01_REINSPORTA_Mcr_01_Oggetti_GEFILE_to_ARchivio _
    02_GetFileName(fullPath As String) As String _
    03_Function GetOggettiDaEsportare() _
    04_Function CancellaOggettoNelDBEsterno(dbPath As String, nomeOggetto As String, tipoOggetto As AcObjectType)
    
    
'//01_REINSPORTA_Mcr_01_Oggetti_GEFILE_to_ARchivio
Public Function REINSPORTA_Mcr_01_Oggetti_GEFILE_to_ARchivio()
    On Error GoTo REINSPORTA_Err
        
        
        Dim dbMadrePath As String
        Dim dbMadreFile As String
        Dim dbCorrenteFile As String
        
        dbMadrePath = "C:\GESTIONI\GESTIONE_LLPP\25_GESTIONE_LLPP\LLPP_ARCHIVI_MDB\"
        dbMadreFile = "MSys_GE_FILE.mdb"
        
        ' Estrai solo il nome file dal database corrente
        dbCorrenteFile = Dir(CurrentDb.Name)
        
        If LCase(dbCorrenteFile) = LCase(dbMadreFile) Then
            Debug.Print "Sono nel database madre: " & CurrentDb.Name
            MsgBox "Sei già nel database madre (MSys_GE_FILE.mdb). Operazione annullata.", vbExclamation
            Exit Function
        End If

        DoCmd.Echo False
        MsgBox "Attendere: esportazione degli oggetti aggiornati in corso nel database sorgente...", vbInformation, "ESPORTAZIONE IN CORSO"

    Dim oggetti As Variant
    oggetti = GetOggettiDaEsportare()

    Dim i As Long
    Dim nomeOggetto As String
    Dim tipoOggetto As AcObjectType

    For i = LBound(oggetti) To UBound(oggetti)
        nomeOggetto = oggetti(i)(0)
        tipoOggetto = oggetti(i)(1)

        ' Per moduli, maschere e query: cancella oggetto nel db esterno se esiste
        If tipoOggetto = acForm Or tipoOggetto = acQuery Or tipoOggetto = acModule Then
            CancellaOggettoNelDBEsterno dbMadrePath, nomeOggetto, tipoOggetto
        End If

        ' Copia l'oggetto nel database esterno
        DoCmd.CopyObject dbMadrePath, nomeOggetto, tipoOggetto, nomeOggetto
    Next i

    MsgBox "Reimportazione completata.", vbInformation

REINSPORTA_Exit:
    DoCmd.Echo True
    Exit Function

REINSPORTA_Err:
    MsgBox "Errore durante la reimportazione: " & Err.Description, vbCritical
    Resume REINSPORTA_Exit
End Function



' Funzione helper per estrarre il nome file da percorso
' 02_GetFileName(fullPath As String)

Private Function GetFileName(fullPath As String) As String
    On Error GoTo ErrHandler

    If Len(Trim(fullPath)) = 0 Then
        Debug.Print "GetFileName: percorso vuoto!"
        GetFileName = ""
        Exit Function
    End If

    GetFileName = Dir(fullPath)
    Debug.Print "GetFileName: estratto nome file -> " & GetFileName
    Exit Function

ErrHandler:
    Debug.Print "GetFileName - Errore: " & Err.Description
    GetFileName = ""
End Function


' 04_Function CancellaOggettoNelDBEsterno(dbPath As String, nomeOggetto As String, tipoOggetto As AcObjectType)
' Funzione che cancella un oggetto (form, query, modulo) nel database esterno usando Access.Application
Private Sub CancellaOggettoNelDBEsterno(dbPath As String, nomeOggetto As String, tipoOggetto As AcObjectType)
    Dim accApp As Object ' Access.Application
    Dim tipoMSys As Long
    Dim rst As DAO.Recordset

    On Error GoTo ErrHandler

    Set accApp = CreateObject("Access.Application")
    accApp.OpenCurrentDatabase dbPath

    Select Case tipoOggetto
        Case acForm: tipoMSys = -32764
        Case acQuery: tipoMSys = 5
        Case acModule: tipoMSys = -32768
        Case Else
            MsgBox "Tipo oggetto non supportato per cancellazione", vbExclamation
            GoTo CleanUp
    End Select

    Set rst = accApp.CurrentDb.OpenRecordset( _
        "SELECT Name FROM MSysObjects WHERE Name='" & nomeOggetto & "' AND Type=" & tipoMSys, dbOpenSnapshot)

    If Not rst.EOF Then
        accApp.DoCmd.DeleteObject tipoOggetto, nomeOggetto
    End If
    rst.Close

CleanUp:
    accApp.Quit
    Set accApp = Nothing
    Exit Sub

ErrHandler:
    MsgBox "Errore durante la cancellazione oggetto: " & Err.Description, vbCritical
    Resume CleanUp
End Sub

        Array("Msys_GEFILE_Qry01_Report", acQuery), _
        Array("frmGestioneFile", acForm), _
        Array("rptStampaDettagli", acReport) _
    )
End Function
